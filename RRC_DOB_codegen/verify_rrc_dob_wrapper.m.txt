function report = verify_rrc_dob_wrapper()
%VERIFY_RRC_DOB_WRAPPER Replay the Simulink wrapper against the integer step.

algorithmDirectory = fileparts(mfilename('fullpath'));
modelName = 'rrc_dob_voltage_compensation_wrapper';
modelFile = fullfile(algorithmDirectory, [modelName '.slx']);
assert(isfile(modelFile), 'RRCDOB:MissingWrapper', ...
    'Missing RRC-DOB wrapper model: %s', modelFile);

sampleTime_s = 50e-6;
sampleCount = 161;
time_s = (0:(sampleCount - 1)).' * sampleTime_s;
electricalFrequency_Hz = 100;
[coefficientsDQ_Q27, q_Q29] = localQuantizedCoefficients( ...
    electricalFrequency_Hz);

phase_rad = 2 * pi * 6 * electricalFrequency_Hz * time_s;
previousAppliedVoltageDQ_Q15 = localToQ15([ ...
    0.15 * sin(phase_rad - 0.15), ...
    0.20 * cos(phase_rad + 0.12)]);
currentDQ_Q15 = localToQ15([ ...
    0.12 * sin(phase_rad), ...
    0.16 * cos(phase_rad + 0.20)]);
rawPiVoltageDQ_Q15 = localToQ15([ ...
    0.18 * sin(phase_rad + 0.30), ...
    0.22 * cos(phase_rad - 0.10)]);
coefficientData = repmat(reshape(coefficientsDQ_Q27, [1, 19, 2]), ...
    sampleCount, 1, 1);
qData = repmat(q_Q29, sampleCount, 1);
resetData = zeros(sampleCount, 1, 'uint8');
rampData = repmat(uint32(2^31), sampleCount, 1);
outputLimitData = repmat(int16(round(0.10 * 2^15)), sampleCount, 1);

inputDataset = Simulink.SimulationData.Dataset;
inputDataset = inputDataset.addElement(timeseries( ...
    previousAppliedVoltageDQ_Q15, time_s, 'IsTimeFirst', true), ...
    'Previous_Applied_Voltage_DQ_Q15');
inputDataset = inputDataset.addElement(timeseries(currentDQ_Q15, time_s, ...
    'IsTimeFirst', true), ...
    'Current_DQ_Q15');
inputDataset = inputDataset.addElement(timeseries(rawPiVoltageDQ_Q15, time_s, ...
    'IsTimeFirst', true), ...
    'Raw_PI_Voltage_DQ_Q15');
inputDataset = inputDataset.addElement(timeseries( ...
    permute(coefficientData, [2, 3, 1]), time_s), ...
    'Coefficients_DQ_Q27');
inputDataset = inputDataset.addElement(timeseries(qData, time_s, ...
    'IsTimeFirst', true), 'Q_Q29');
inputDataset = inputDataset.addElement(timeseries(resetData, time_s, ...
    'IsTimeFirst', true), ...
    'Reset_u8');
inputDataset = inputDataset.addElement(timeseries(rampData, time_s, ...
    'IsTimeFirst', true), ...
    'Ramp_Q31');
inputDataset = inputDataset.addElement(timeseries(outputLimitData, time_s, ...
    'IsTimeFirst', true), ...
    'Output_Limit_Q15');

simulationInput = Simulink.SimulationInput(modelName);
simulationInput = simulationInput.setModelParameter( ...
    'StopTime', num2str(time_s(end), 17), ...
    'SaveOutput', 'on', ...
    'OutputSaveName', 'yout', ...
    'SaveFormat', 'Dataset');
simulationInput = simulationInput.setExternalInput(inputDataset);

originalDirectory = pwd;
directoryCleanup = onCleanup(@() cd(originalDirectory));
cd(algorithmDirectory);
load_system(modelFile);
simulationOutput = sim(simulationInput);
clear directoryCleanup;

commandExpected = zeros(sampleCount, 2, 'int16');
estimateExpected = zeros(sampleCount, 2, 'int16');
stateExpected = zeros(sampleCount, 8, 'int32');
statusExpected = zeros(sampleCount, 1, 'uint8');
state = zeros(8, 1, 'int32');
for sampleIndex = 1:sampleCount
    [command, estimate, state, status] = rrcDobFixedPointStep( ...
        state, previousAppliedVoltageDQ_Q15(sampleIndex, :).', ...
        currentDQ_Q15(sampleIndex, :).', ...
        rawPiVoltageDQ_Q15(sampleIndex, :).', coefficientsDQ_Q27, ...
        q_Q29, resetData(sampleIndex), rampData(sampleIndex), ...
        outputLimitData(sampleIndex));
    commandExpected(sampleIndex, :) = command.';
    estimateExpected(sampleIndex, :) = estimate.';
    stateExpected(sampleIndex, :) = state.';
    statusExpected(sampleIndex) = status;
end

outputDataset = simulationOutput.get('yout');
commandActual = localDatasetMatrix(outputDataset, 'Command_DQ_Q15', 1, ...
    sampleCount, 2, 'int16');
estimateActual = localDatasetMatrix(outputDataset, ...
    'Error_Hat_DQ_Q15', 2, sampleCount, 2, 'int16');
stateActual = localDatasetMatrix(outputDataset, 'State_Q26', 3, ...
    sampleCount, 8, 'int32');
statusActual = localDatasetMatrix(outputDataset, 'Status_u8', 4, ...
    sampleCount, 1, 'uint8');

assert(isequal(commandActual, commandExpected), ...
    'RRCDOB:WrapperCommandMismatch', ...
    'Wrapper command output differs from the integer reference.');
assert(isequal(estimateActual, estimateExpected), ...
    'RRCDOB:WrapperEstimateMismatch', ...
    'Wrapper estimate output differs from the integer reference.');
assert(isequal(stateActual, stateExpected), ...
    'RRCDOB:WrapperStateMismatch', ...
    'Wrapper state output differs from the integer reference.');
assert(isequal(statusActual, statusExpected), ...
    'RRCDOB:WrapperStatusMismatch', ...
    'Wrapper status output differs from the integer reference.');

report = struct( ...
    'sampleCount', sampleCount, ...
    'commandBitExact', true, ...
    'estimateBitExact', true, ...
    'stateBitExact', true, ...
    'statusBitExact', true, ...
    'passed', true);
fprintf('RRC-DOB wrapper replay: %d samples, bit-exact PASS.\n', ...
    sampleCount);
end

function values = localDatasetMatrix(dataset, elementName, elementIndex, ...
        sampleCount, signalWidth, className)
element = dataset.getElement(elementIndex);
assert(~isempty(element), 'RRCDOB:MissingWrapperOutput', ...
    'Wrapper output %s is missing.', elementName);
rawValues = element.Values.Data;
if ndims(rawValues) == 3 && size(rawValues, 3) == sampleCount
    rawValues = permute(rawValues, [3, 1, 2]);
end
values = reshape(rawValues, sampleCount, signalWidth);
values = cast(values, className);
end

function [coefficientsDQ_Q27, q_Q29] = localQuantizedCoefficients( ...
        electricalFrequency_Hz)
resistance_Ohm = 0.5;
inductances_H = [1.30e-3, 1.38e-3];
voltageBase_V = 1000;
currentBase_A = 50;
sampleTime_s = 50e-6;
cutoffRatio = 0.10;
halfSampleTime_s = sampleTime_s / 2;
q = tan(2 * pi * 6 * electricalFrequency_Hz * halfSampleTime_s);
q_Q29 = int32(round(q * 2^29));
coefficientsDQ_Q27 = zeros(19, 2, 'int32');

for axisIndex = 1:2
    inductance_H = inductances_H(axisIndex);
    qSquared = q * q;
    twoCutoffQ = 2 * cutoffRatio * q;
    r = resistance_Ohm * halfSampleTime_s / inductance_H;
    onePlusR = 1 + r;
    denominator = 1 + twoCutoffQ + qSquared;
    commonDenominator = onePlusR * denominator;
    voltageInputGain = halfSampleTime_s * voltageBase_V / ...
        (inductance_H * currentBase_A);
    lambda = inductance_H * currentBase_A / ...
        (voltageBase_V * sampleTime_s);
    rho = resistance_Ohm * currentBase_A / voltageBase_V;
    coefficients = [ ...
        ((1-r)*(1+qSquared)-twoCutoffQ*onePlusR)/commonDenominator; ...
        -4*cutoffRatio*q/denominator; ...
        -4*cutoffRatio*(qSquared-r)/commonDenominator; ...
        -2*qSquared/commonDenominator; ...
        (1+twoCutoffQ-qSquared)/denominator; ...
        2*q*(onePlusR+twoCutoffQ)/commonDenominator; ...
        -2*q/commonDenominator; -2*q/denominator; ...
        (onePlusR*(1-qSquared)+twoCutoffQ*(1-r))/commonDenominator; ...
        2*voltageInputGain*(1+qSquared)/commonDenominator; ...
        4*cutoffRatio*q/denominator; ...
        -2*voltageInputGain*qSquared/commonDenominator; ...
        2*qSquared/denominator; ...
        -2*voltageInputGain*q/commonDenominator; 2*q/denominator; ...
        -4*cutoffRatio*q*lambda; -4*cutoffRatio*q*lambda; ...
        2*cutoffRatio*rho; 4*cutoffRatio*q*lambda];
    coefficientsDQ_Q27(:, axisIndex) = int32(round(coefficients * 2^27));
end
end

function valuesQ15 = localToQ15(values)
wideValues = round(values * 2^15);
wideValues = min(max(wideValues, double(intmin('int16'))), ...
    double(intmax('int16')));
valuesQ15 = int16(wideValues);
end
