function report = verify_apsfsm_torque_compensation(sourceVectorFile)
%verify_apsfsm_torque_compensation Run MATLAB reference and source replay.
%
% The checked-in MAT file contains the 7801 real samples extracted from the
% source task. A different file can be supplied for independent replay, but
% it must expose the same fields documented in README.md.

thisDirectory = fileparts(mfilename('fullpath'));
if nargin < 1
    sourceVectorFile = fullfile(thisDirectory, 'testdata', ...
        'apsfsm_source_replay_vectors.mat');
end

suite = matlab.unittest.TestSuite.fromFile(fullfile(thisDirectory, ...
    'apsfsmTorqueCompensationStepTest.m'));
testResults = run(suite);

replay = load(sourceVectorFile);
source = replay.source;
parameters = single([source.omegaBase_radps; source.kHat; ...
    source.rho_rad; source.lambda; source.coefficientLimit_PU]);
state = zeros(4, 1, 'single');
state(4) = single(0.5) * parameters(2) * parameters(2);
iqRaw = zeros(size(replay.speedFeedback_PU), 'single');

for sampleIndex = 1:numel(iqRaw)
    publishedState = state;
    learnEnable = single(replay.learnEnable_u8(sampleIndex));
    control = single([learnEnable; 0; 1 - learnEnable]);
    [state, diagnostics] = apsfsmTorqueCompensationStep( ...
        state, replay.speedReference_PU(sampleIndex), ...
        replay.speedFeedback_PU(sampleIndex), parameters, control);
    iqRaw(sampleIndex) = diagnostics(1);
end

terminalHarmonicPeak = hypot(publishedState(1), publishedState(2));
absoluteErrors = double([ ...
    abs(publishedState(1) - single(source.goldenBHat_PU)); ...
    abs(publishedState(2) - single(source.goldenCHat_PU)); ...
    abs(terminalHarmonicPeak - single(source.goldenIqPeak_PU))]);
sourceTolerance = 1.0e-4;
sourcePassed = all(absoluteErrors <= sourceTolerance) && ...
    all(isfinite(publishedState)) && all(isfinite(state)) && ...
    all(isfinite(iqRaw));

report = struct();
report.testResults = testResults;
report.sourceReplay = struct( ...
    'vectorFile', string(sourceVectorFile), ...
    'sampleCount', numel(iqRaw), ...
    'terminalBHatPU', double(publishedState(1)), ...
    'terminalCHatPU', double(publishedState(2)), ...
    'terminalHarmonicPeakPU', double(terminalHarmonicPeak), ...
    'postFinalUpdateState', double(state), ...
    'maximumTransientIqRawPU', double(max(abs(iqRaw))), ...
    'absoluteErrorsPU', absoluteErrors, ...
    'tolerancePU', sourceTolerance, ...
    'allFinite', all(isfinite(publishedState)) && ...
        all(isfinite(state)) && all(isfinite(iqRaw)), ...
    'passed', sourcePassed);
report.passed = all([testResults.Passed]) && sourcePassed;

fprintf('MATLAB tests: %d passed, %d failed, %d incomplete.\n', ...
    nnz([testResults.Passed]), nnz([testResults.Failed]), ...
    nnz([testResults.Incomplete]));
fprintf(['Source replay: B=%.9f PU, C=%.9f PU, terminal peak=' ...
    '%.9f PU, max error=%.3g PU (%s).\n'], ...
    report.sourceReplay.terminalBHatPU, ...
    report.sourceReplay.terminalCHatPU, ...
    report.sourceReplay.terminalHarmonicPeakPU, max(absoluteErrors), ...
    string(localPassLabel(sourcePassed)));
end

function label = localPassLabel(passed)
if passed
    label = 'PASS';
else
    label = 'FAIL';
end
end
