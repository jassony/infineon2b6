function [fadoStateNext, fadoOutput] = fado_external_observer_discrete_update( ...
    fadoEnable, fadoVoltageAlpha_V, fadoVoltageBeta_V, ...
    fadoCurrentAlpha_A, fadoCurrentBeta_A, ...
    fadoStatorResistance_Ohm, fadoQuadratureInductance_H, fadoPolePairs, ...
    fadoFluxLimit_Wb, fadoKdf_per_s, fadoLowSpeedThreshold_Hz, ...
    fadoKaf_radps, fadoFastT2SBandwidth_Hz, fadoSlowT2SBandwidth_Hz, ...
    fadoT2SDamping, fadoVoltageAlphaOffset_V, fadoVoltageBetaOffset_V, ...
    fadoVoltagePreprocessEnable, fadoState)
%#codegen
% Explicit-Euler FADO update. The state is supplied by a Unit Delay block.

fadoTs_s = single(5.0e-5);
fadoTwoPi = single(6.283185307179586);
fadoStateNext = zeros(10, 1, 'single');
fadoOutput = zeros(18, 1, 'single');

if ~fadoEnable
    return;
end

fadoOutput(18) = single(1.0);
if ~localCalibrationIsValid(fadoStatorResistance_Ohm, ...
        fadoQuadratureInductance_H, fadoPolePairs, fadoFluxLimit_Wb, ...
        fadoKdf_per_s, fadoLowSpeedThreshold_Hz, fadoKaf_radps, ...
        fadoFastT2SBandwidth_Hz, fadoSlowT2SBandwidth_Hz, fadoT2SDamping, ...
        fadoVoltageAlphaOffset_V, fadoVoltageBetaOffset_V)
    fadoOutput(17) = single(3.0);
    return;
end

fadoLambdaAlpha1_Wb = fadoState(1);
fadoLambdaBeta1_Wb = fadoState(2);
fadoLambdaAlphaHat_Wb = fadoState(3);
fadoLambdaBetaHat_Wb = fadoState(4);
fadoDhatAlpha_Wb = fadoState(5);
fadoDhatBeta_Wb = fadoState(6);
fadoThetaFast_rad = fadoState(7);
fadoThetaSlow_rad = fadoState(8);
fadoOmegaFast_radps = fadoState(9);
fadoOmegaSlow_radps = fadoState(10);

if fadoVoltagePreprocessEnable
    fadoVoltageAlpha_V = fadoVoltageAlpha_V + fadoVoltageAlphaOffset_V;
    fadoVoltageBeta_V = fadoVoltageBeta_V + fadoVoltageBetaOffset_V;
end

fadoOmegaThreshold_radps = fadoTwoPi * fadoLowSpeedThreshold_Hz;
fadoKafMode = abs(fadoOmegaFast_radps) <= fadoOmegaThreshold_radps;
fadoKdfMode = ~fadoKafMode;
fadoFluxLimited = false;

fadoRotorFluxAlpha_Wb = fadoLambdaAlpha1_Wb - fadoDhatAlpha_Wb ...
    - fadoQuadratureInductance_H * fadoCurrentAlpha_A;
fadoRotorFluxBeta_Wb = fadoLambdaBeta1_Wb - fadoDhatBeta_Wb ...
    - fadoQuadratureInductance_H * fadoCurrentBeta_A;
fadoRotorFluxMag_Wb = sqrt(fadoRotorFluxAlpha_Wb * fadoRotorFluxAlpha_Wb ...
    + fadoRotorFluxBeta_Wb * fadoRotorFluxBeta_Wb);

if fadoKafMode
    fadoRotorFluxAlphaLimited_Wb = fadoRotorFluxAlpha_Wb;
    fadoRotorFluxBetaLimited_Wb = fadoRotorFluxBeta_Wb;
    if fadoRotorFluxMag_Wb > fadoFluxLimit_Wb
        fadoLimiterScale = fadoFluxLimit_Wb / fadoRotorFluxMag_Wb;
        fadoRotorFluxAlphaLimited_Wb = fadoRotorFluxAlpha_Wb * fadoLimiterScale;
        fadoRotorFluxBetaLimited_Wb = fadoRotorFluxBeta_Wb * fadoLimiterScale;
        fadoFluxLimited = true;
    end
    fadoLambdaAlphaDot_V = fadoVoltageAlpha_V ...
        - fadoStatorResistance_Ohm * fadoCurrentAlpha_A ...
        - fadoKaf_radps * (fadoRotorFluxAlpha_Wb - fadoRotorFluxAlphaLimited_Wb);
    fadoLambdaBetaDot_V = fadoVoltageBeta_V ...
        - fadoStatorResistance_Ohm * fadoCurrentBeta_A ...
        - fadoKaf_radps * (fadoRotorFluxBeta_Wb - fadoRotorFluxBetaLimited_Wb);
    fadoKdfEffective_per_s = single(0.0);
else
    fadoKdfEffective_per_s = fadoKdf_per_s;
    fadoLambdaAlphaDot_V = fadoVoltageAlpha_V ...
        - fadoStatorResistance_Ohm * fadoCurrentAlpha_A ...
        - fadoKdfEffective_per_s * fadoDhatAlpha_Wb;
    fadoLambdaBetaDot_V = fadoVoltageBeta_V ...
        - fadoStatorResistance_Ohm * fadoCurrentBeta_A ...
        - fadoKdfEffective_per_s * fadoDhatBeta_Wb;
end

fadoLambdaAlpha1_Wb = fadoLambdaAlpha1_Wb + fadoTs_s * fadoLambdaAlphaDot_V;
fadoLambdaBeta1_Wb = fadoLambdaBeta1_Wb + fadoTs_s * fadoLambdaBetaDot_V;

fadoAbsOmega_radps = abs(fadoOmegaFast_radps);
fadoInnovationAlpha_Wb = fadoLambdaAlpha1_Wb - fadoLambdaAlphaHat_Wb;
fadoInnovationBeta_Wb = fadoLambdaBeta1_Wb - fadoLambdaBetaHat_Wb;
fadoLambdaAlphaHatDot_V = -fadoOmegaFast_radps * fadoLambdaBetaHat_Wb ...
    + fadoOmegaFast_radps * fadoDhatBeta_Wb ...
    + single(2.0) * fadoAbsOmega_radps * fadoInnovationAlpha_Wb ...
    - fadoOmegaFast_radps * fadoInnovationBeta_Wb;
fadoLambdaBetaHatDot_V = fadoOmegaFast_radps * fadoLambdaAlphaHat_Wb ...
    - fadoOmegaFast_radps * fadoDhatAlpha_Wb ...
    + fadoOmegaFast_radps * fadoInnovationAlpha_Wb ...
    + single(2.0) * fadoAbsOmega_radps * fadoInnovationBeta_Wb;
fadoDhatAlphaDot_V = -fadoOmegaFast_radps * fadoInnovationBeta_Wb;
fadoDhatBetaDot_V = fadoOmegaFast_radps * fadoInnovationAlpha_Wb;

fadoLambdaAlphaHat_Wb = fadoLambdaAlphaHat_Wb + fadoTs_s * fadoLambdaAlphaHatDot_V;
fadoLambdaBetaHat_Wb = fadoLambdaBetaHat_Wb + fadoTs_s * fadoLambdaBetaHatDot_V;
fadoDhatAlpha_Wb = fadoDhatAlpha_Wb + fadoTs_s * fadoDhatAlphaDot_V;
fadoDhatBeta_Wb = fadoDhatBeta_Wb + fadoTs_s * fadoDhatBetaDot_V;

fadoLambdaAlpha2_Wb = fadoLambdaAlpha1_Wb - fadoDhatAlpha_Wb;
fadoLambdaBeta2_Wb = fadoLambdaBeta1_Wb - fadoDhatBeta_Wb;
fadoThetaDirect_rad = atan2(fadoLambdaBeta2_Wb ...
    - fadoQuadratureInductance_H * fadoCurrentBeta_A, ...
    fadoLambdaAlpha2_Wb - fadoQuadratureInductance_H * fadoCurrentAlpha_A);

fadoFastNaturalFrequency_radps = fadoTwoPi * fadoFastT2SBandwidth_Hz;
fadoFastPhaseError_rad = localWrapToPi(fadoThetaDirect_rad - fadoThetaFast_rad);
fadoOmegaFast_radps = fadoOmegaFast_radps + fadoTs_s ...
    * fadoFastNaturalFrequency_radps * fadoFastNaturalFrequency_radps * fadoFastPhaseError_rad;
fadoThetaFast_rad = localWrapToPi(fadoThetaFast_rad + fadoTs_s ...
    * (fadoOmegaFast_radps + single(2.0) * fadoT2SDamping ...
    * fadoFastNaturalFrequency_radps * fadoFastPhaseError_rad));

fadoSlowNaturalFrequency_radps = fadoTwoPi * fadoSlowT2SBandwidth_Hz;
fadoSlowPhaseError_rad = localWrapToPi(fadoThetaDirect_rad - fadoThetaSlow_rad);
fadoOmegaSlow_radps = fadoOmegaSlow_radps + fadoTs_s ...
    * fadoSlowNaturalFrequency_radps * fadoSlowNaturalFrequency_radps * fadoSlowPhaseError_rad;
fadoThetaSlow_rad = localWrapToPi(fadoThetaSlow_rad + fadoTs_s ...
    * (fadoOmegaSlow_radps + single(2.0) * fadoT2SDamping ...
    * fadoSlowNaturalFrequency_radps * fadoSlowPhaseError_rad));

fadoStateNext(1) = fadoLambdaAlpha1_Wb;
fadoStateNext(2) = fadoLambdaBeta1_Wb;
fadoStateNext(3) = fadoLambdaAlphaHat_Wb;
fadoStateNext(4) = fadoLambdaBetaHat_Wb;
fadoStateNext(5) = fadoDhatAlpha_Wb;
fadoStateNext(6) = fadoDhatBeta_Wb;
fadoStateNext(7) = fadoThetaFast_rad;
fadoStateNext(8) = fadoThetaSlow_rad;
fadoStateNext(9) = fadoOmegaFast_radps;
fadoStateNext(10) = fadoOmegaSlow_radps;

if ~all(isfinite(fadoStateNext))
    fadoStateNext(:) = single(0.0);
    fadoOutput(17) = single(2.0);
    return;
end

fadoOutput(1) = mod(fadoThetaFast_rad, fadoTwoPi);
fadoOutput(2) = fadoOmegaFast_radps;
fadoOutput(3) = fadoOmegaSlow_radps;
fadoOutput(4) = fadoOmegaSlow_radps / (fadoTwoPi * single(fadoPolePairs)) * single(60.0);
fadoOutput(5) = fadoLambdaAlpha1_Wb;
fadoOutput(6) = fadoLambdaBeta1_Wb;
fadoOutput(7) = fadoLambdaAlpha2_Wb;
fadoOutput(8) = fadoLambdaBeta2_Wb;
fadoOutput(9) = fadoDhatAlpha_Wb;
fadoOutput(10) = fadoDhatBeta_Wb;
fadoOutput(11) = fadoRotorFluxMag_Wb;
fadoOutput(12) = fadoKdfEffective_per_s;
fadoOutput(13) = single(fadoKdfMode);
fadoOutput(14) = single(fadoKafMode);
fadoOutput(15) = single(fadoFluxLimited);
fadoOutput(16) = single(1.0);
fadoOutput(17) = single(1.0);
end

function fadoAngleWrapped_rad = localWrapToPi(fadoAngle_rad)
fadoPi = single(3.141592653589793);
fadoTwoPi = single(6.283185307179586);
fadoAngleWrapped_rad = mod(fadoAngle_rad + fadoPi, fadoTwoPi) - fadoPi;
end

function isValid = localCalibrationIsValid(fadoStatorResistance_Ohm, ...
    fadoQuadratureInductance_H, fadoPolePairs, fadoFluxLimit_Wb, ...
    fadoKdf_per_s, fadoLowSpeedThreshold_Hz, fadoKaf_radps, ...
    fadoFastT2SBandwidth_Hz, fadoSlowT2SBandwidth_Hz, fadoT2SDamping, ...
    fadoVoltageAlphaOffset_V, fadoVoltageBetaOffset_V)

isValid = isfinite(fadoStatorResistance_Ohm) ...
    && isfinite(fadoQuadratureInductance_H) ...
    && isfinite(fadoFluxLimit_Wb) ...
    && isfinite(fadoKdf_per_s) ...
    && isfinite(fadoLowSpeedThreshold_Hz) ...
    && isfinite(fadoKaf_radps) ...
    && isfinite(fadoFastT2SBandwidth_Hz) ...
    && isfinite(fadoSlowT2SBandwidth_Hz) ...
    && isfinite(fadoT2SDamping) ...
    && isfinite(fadoVoltageAlphaOffset_V) ...
    && isfinite(fadoVoltageBetaOffset_V) ...
    && (fadoStatorResistance_Ohm > single(0.0)) ...
    && (fadoQuadratureInductance_H > single(0.0)) ...
    && (fadoPolePairs > uint8(0)) ...
    && (fadoFluxLimit_Wb > single(0.0)) ...
    && (fadoKdf_per_s >= single(0.0)) ...
    && (fadoLowSpeedThreshold_Hz >= single(0.0)) ...
    && (fadoKaf_radps > single(0.0)) ...
    && (fadoFastT2SBandwidth_Hz > single(0.0)) ...
    && (fadoSlowT2SBandwidth_Hz > single(0.0)) ...
    && (fadoT2SDamping > single(0.0));
end
