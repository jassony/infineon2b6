function dv_export(derivation, realization, parameterSymbols, outDir)
%DV_EXPORT Export symbolic/numeric recurrences and a parameter coefficient file.
% Writes only to the explicitly supplied output directory.
if ~isfolder(outDir), mkdir(outDir); end
method=derivation.method;
coefficientFile=fullfile(outDir,['dv_coefficients_' method '.m']);
matlabFunction(derivation.Ad,derivation.Bd,derivation.Cd,derivation.Dd, ...
    'File',coefficientFile,'Vars',num2cell(parameterSymbols), ...
    'Outputs',{'Ad','Bd','Cd','Dd'},'Optimize',true);
fid=fopen(fullfile(outDir,'difference_equations.md'),'w','n','UTF-8');
assert(fid>=0,'dv:Export','Cannot open difference-equation report.');
cleanup=onCleanup(@() fclose(fid)); %#ok<NASGU>
fprintf(fid,'# %s discretization\n\n',method);
fprintf(fid,'Ts = %.17g s. Coefficient generator inputs, in order: `%s`.\n\n', ...
    realization.Ts,strjoin(string(parameterSymbols),', '));
fprintf(fid,'Conditions: %s.\n\nState: %s.\n\n',derivation.conditions,derivation.stateMeaning);
fprintf(fid,'Parameter values in the same order: `%s`.\n\n',mat2str(realization.parameterValues,17));
fprintf(fid,'Symbolic proof: %s. Result: `%s` (empty means unproven).\n\n', ...
    derivation.symbolicProofKind,mat2str(derivation.symbolicIdentity));
if ~isempty(derivation.Gc)
    fprintf(fid,'## Transfer functions\n\n```text\nGc(s) = %s\nGd(z) = %s\n```\n\n', ...
        char(derivation.Gc),char(derivation.Gd));
end
fprintf(fid,['`xi[k+1] = Ad*xi[k] + Bd*u[k]`; `y[k] = Cd*xi[k] + Dd*u[k]`.\n\n' ...
    'Compute y from the old state before assigning xiNext. Reset clears state and output, ' ...
    'including direct feedthrough, and ignores that sample input. Parameters are frozen.\n\n' ...
    'Default initial xi is zero (zero prehistory). This need not mean physical x[0]=0 ' ...
    'for BE/Tustin when u[0] is nonzero. For a physical initial condition use ' ...
    '`xi0 = initialStateMatrix*x0 + initialInputMatrix*u0` and validate separately.\n\n']);
for field={'A','B','C','D','Ad','Bd','Cd','Dd','initialStateMatrix','initialInputMatrix'}
    name=field{1};
    fprintf(fid,'## %s\n\nSymbolic:\n```text\n%s\n```\nNumeric:\n```text\n%s\n```\n\n', ...
        name,char(derivation.(name)),mat2str(realization.(name),17));
end
if ~isempty(derivation.recurrence)
fprintf(fid,['## Input/output difference equations\n\n' ...
    '`q = z^-1`. Each row below represents one input/output channel v_ij. ' ...
    '`y_i[k] = sum_j v_ij[k]`. For SISO v_11 is y.\n\n' ...
    '`v_ij[k] = -sum(a(l+1)*v_ij[k-l], l=1..na) + ' ...
    'sum(b(l+1)*u_j[k-l], l=0..nb)`.\n\n' ...
    '`a(1)=1`; b(1) is the current-input term and must not be dropped. ' ...
    'Initialize and reset all per-channel input/output histories to zero.\n\n']);
for iy=1:size(derivation.recurrence,1)
    for iu=1:size(derivation.recurrence,2)
        symRow=derivation.recurrence{iy,iu}; numRow=realization.recurrence{iy,iu};
        ioFile=fullfile(outDir,sprintf('dv_io_coefficients_%s_o%d_i%d.m',method,iy,iu));
        matlabFunction(symRow.b,symRow.a,'File',ioFile, ...
            'Vars',num2cell(parameterSymbols),'Outputs',{'b','a'},'Optimize',true);
        fprintf(fid,'### Channel %d <- %d\n\n```text\na(q): %s\nb(q): %s\na_numeric: %s\nb_numeric: %s\n```\n\n', ...
            iy,iu,char(symRow.a),char(symRow.b),mat2str(numRow.a,17),mat2str(numRow.b,17));
    end
end
end
fprintf(fid,['For high order/poorly conditioned polynomials prefer the state recurrence or a separately ' ...
    'validated SOS realization. Cancellation in a channel TF does not remove hidden states ' ...
    'from the SS stability check. Generated files are MATLAB references, not code-generation acceptance.\n']);
end
