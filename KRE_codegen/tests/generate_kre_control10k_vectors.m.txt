function result = generate_kre_control10k_vectors()
% Deterministic 20 kHz held-voltage plant / 10 kHz observer reference.
% Numerical tolerances are fixed here before the C replay is run. This is
% synthetic replay, not proof of hardware sampling or closed-loop stability.
root = fileparts(fileparts(fileparts(mfilename('fullpath'))));
addpath(fullfile(root,'KRE_codegen'));
out = fullfile(root,'Build','Control10kTests');
if ~isfolder(out), mkdir(out); end
fid = fopen(fullfile(out,'kre_control10k_vectors.h'),'w');
assert(fid >= 0);
cleanup = onCleanup(@() fclose(fid)); %#ok<NASGU>
fprintf(fid,'/* Generated by generate_kre_control10k_vectors.m */\n');
fprintf(fid,'#define REPLAY_ANGLE_TOL 0.002F\n#define REPLAY_SPEED_TOL 2.0F\n');
fprintf(fid,'#define REPLAY_FLUX_TOL 0.00005F\n#define REPLAY_OMEGA_TOL 1.0F\n');
fprintf(fid,'static const float replay[][9] = {\n');
ts = single(100)*single(1e-6);
params = single([0.5 0 0 0 ts 1000 50 10000 4 1256.637061 125.663704 1 1e-5 150]);
params(2) = single(1300)*single(1e-6);
params(3) = single(1380)*single(1e-6);
params(4) = single(46)*single(0.001);
pll = single([100 0.70710678]);
rows = 0;
for scenario = 1:3
    clear kre_external_observer_diagnostic_step
    delayed = zeros(4,1,'single');
    current = zeros(2,1,'single');
    heldVoltage = zeros(2,1,'single');
    theta = single(0);
    for n = 1:600
        reset = n == 1 || n == 301;
        if reset
            clear kre_external_observer_diagnostic_step
            delayed(:) = 0;
        end
        % Stop, reverse and varying frequency cases. One voltage command is
        % held for TWO 50 us PWM plant steps before the next control sample.
        freq = single(60 + 20*(scenario==3));
        if scenario == 2, freq = -freq; end
        if n > 400, freq = single(0); end
        omega = single(2*pi)*freq;
        for pwm = 1:2
            theta = theta + single(50e-6)*omega;
            emf = params(4)*omega*[-sin(theta);cos(theta)];
            current = current + single(50e-6)*(heldVoltage-params(1)*current-emf)/params(2);
        end
        measured = [heldVoltage;current];
        [pos,speed,~,flux,raw,status] = kre_external_observer_diagnostic_step(uint8(1),delayed,params,pll);
        expected = [pos*single(2*pi); speed*params(8); flux; raw];
        assert(status == 1 && all(isfinite(expected)));
        fprintf(fid,'{%.9eF,%.9eF,%.9eF,%.9eF,%.9eF,%.9eF,%.9eF,%.9eF,%u},\n', ...
            double([measured;expected]),reset);
        delayed = [measured(1:2)/params(6);measured(3:4)/params(7)];
        % New command is computed AFTER capturing the voltage that produced
        % the measured currents. It is not passed to this observer sample.
        heldVoltage = params(4)*omega*[-sin(theta);cos(theta)] ...
            + single(2)*[cos(theta);sin(theta)];
        rows = rows + 1;
    end
end
fprintf(fid,'};\n');
result = struct('rows',rows,'pwm_us',50,'control_us',100, ...
    'angleTolerance_rad',0.002,'speedTolerance_rpm',2, ...
    'fluxTolerance_Wb',5e-5,'omegaTolerance_radps',1);
disp(result);
end
