function manifestRow = powerfit_manifest_row(source, status, reason, details, profile, cfg)
%POWERFIT_MANIFEST_ROW Build one typed manifest row with safe defaults.

if nargin < 4 || isempty(details)
    details = struct();
end

manifestRow = table( ...
    string(source.SourceId), string(source.SourceHashSHA256), ...
    string(source.SourceFile), string(source.TestId), ...
    double(source.FileBytes), source.ModifiedTime, ...
    localText(details, "MdfVersion"), localDateTime(details, "MdfInitialTimestamp"), ...
    string(profile.FitGroupId), string(profile.HardwareConfigId), ...
    string(profile.FirmwareId), string(profile.MotorId), string(profile.InverterId), ...
    double(profile.SwitchingFrequency_Hz), double(profile.CoolantTemperature_C), ...
    logical(profile.SupplyVoltageScaleConfirmed), ...
    logical(profile.SupplyCurrentPolarityConfirmed), ...
    logical(profile.DqCurrentMatchesFirmware), ...
    double(cfg.SupplyVoltageScale_V_per_value), ...
    double(cfg.SupplyCurrentScale_A_per_value), ...
    localNumber(details, "DqGroup"), localNumber(details, "DcVoltageGroup"), ...
    localNumber(details, "DcCurrentGroup"), localNumber(details, "FocStateGroup"), ...
    localNumber(details, "FocVdcGroup"), localNumber(details, "SpeedGroup"), ...
    localNumber(details, "RowsRead"), localNumber(details, "EligibleRows"), ...
    localNumber(details, "SupplyVsFocVdcMean_V"), ...
    localNumber(details, "SupplyVsFocVdcMeanAbs_V"), ...
    localNumber(details, "SupplyVsFocVdcMaxAbs_V"), ...
    string(status), string(reason), datetime("now"), ...
    'VariableNames', { ...
    'SourceId', 'SourceHashSHA256', 'SourceFile', 'TestId', ...
    'FileBytes', 'ModifiedTime', 'MdfVersion', 'MdfInitialTimestamp', ...
    'FitGroupId', 'HardwareConfigId', 'FirmwareId', 'MotorId', 'InverterId', ...
    'SwitchingFrequency_Hz', 'CoolantTemperature_C', ...
    'SupplyVoltageScaleConfirmed', 'SupplyCurrentPolarityConfirmed', ...
    'DqCurrentMatchesFirmware', 'SupplyVoltageScale_V_per_value', ...
    'SupplyCurrentScale_A_per_value', ...
    'DqGroup', 'DcVoltageGroup', 'DcCurrentGroup', 'FocStateGroup', ...
    'FocVdcGroup', 'SpeedGroup', 'RowsRead', 'EligibleRows', ...
    'SupplyVsFocVdcMean_V', 'SupplyVsFocVdcMeanAbs_V', 'SupplyVsFocVdcMaxAbs_V', ...
    'Status', 'Reason', 'ImportedAt'});
end

function value = localText(details, name)
value = "";
if isfield(details, name)
    value = string(details.(name));
end
end

function value = localDateTime(details, name)
value = NaT;
if isfield(details, name) && ~isempty(details.(name))
    value = details.(name);
    if isdatetime(value) && ~isempty(value.TimeZone)
        value.TimeZone = '';
    end
end
end

function value = localNumber(details, name)
value = NaN;
if isfield(details, name) && ~isempty(details.(name))
    value = double(details.(name));
end
end
